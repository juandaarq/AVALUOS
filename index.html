Conecta tu hoja de cálculo como base de datos

📋 Instrucciones de Configuración

  1. 1
    Crea una Google Sheet nueva. En la primera fila (fila 1) deja los encabezados exactamente como los genera la app automáticamente.
  2. 2
    Ve a Extensiones → Apps Script y pega el script de la pestaña "Script" en el panel.
  3. 3
    Haz clic en Implementar → Nueva implementación → Aplicación web. En "¿Quién tiene acceso?", selecciona Cualquier usuario. Copia la URL generada.
  4. 4
    Pega la URL del Web App y el ID de tu hoja (el código largo de la URL de Google Sheets) en los campos de abajo.
const SHEET_NAME = "Avaluos";

function doGet(e) {
  try {
    const action = e.parameter.action;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    
    if (action === "getAll") {
      const data = sheet.getDataRange().getValues();
      if (data.length <= 1) return jsonResponse({ records: [] });
      const headers = data[0];
      const records = data.slice(1).map(row => {
        const obj = {};
        headers.forEach((h, i) => obj[h] = row[i]);
        return obj;
      });
      return jsonResponse({ records });
    }
  } catch(err) {
    return jsonResponse({ error: err.message });
  }
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    
    const HEADERS = ["id","sector","soilClass","buildState","area","price","constructionValue","lat","lng","sourceUrl"];
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(HEADERS);
    }
    
    if (action === "add") {
      const row = payload.record;
      sheet.appendRow(HEADERS.map(h => row[h] ?? ""));
      return jsonResponse({ success: true });
    }
    
    if (action === "update") {
      const row = payload.record;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(row.id)) {
          HEADERS.forEach((h, j) => sheet.getRange(i+1, j+1).setValue(row[h] ?? ""));
          return jsonResponse({ success: true });
        }
      }
      return jsonResponse({ error: "ID no encontrado" });
    }
    
    if (action === "delete") {
      const id = payload.id;
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(id)) {
          sheet.deleteRow(i+1);
          return jsonResponse({ success: true });
        }
      }
      return jsonResponse({ error: "ID no encontrado" });
    }
    
    if (action === "seed") {
      sheet.clearContents();
      sheet.appendRow(HEADERS);
      payload.records.forEach(row => {
        sheet.appendRow(HEADERS.map(h => row[h] ?? ""));
      });
      return jsonResponse({ success: true, seeded: payload.records.length });
    }
    
  } catch(err) {
    return jsonResponse({ error: err.message });
  }
}

function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
